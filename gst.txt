#!/bin/bash
# gst.txt — The Scorecard, two clean-ups (prepared by Claude for Willy, 6 Nov 2026)
#   Step 13: old guest sign-ins (no email) that belong to no group and have
#            not been used for 30 days.
#   Step 29: accounts with an email that are missing from the account-email list.
#
#   bash gst.txt       LIST ONLY. Changes nothing. The full list goes to a file;
#                      the screen shows a short summary, last, so it stays visible.
#   bash gst.txt go    Does exactly what the list shows, after saving a backup:
#                      the guests' own small records in ONE commit (all or
#                      nothing), then the guest sign-ins; the missing
#                      account-email entries in ONE commit.
set -u
gcloud config set project scorecard-f41b8 >/dev/null 2>&1
P=scorecard-f41b8
MODE="${1:-list}"
TOKEN="${GST_TOKEN:-$(gcloud auth print-access-token 2>/dev/null)}"
[ -n "$TOKEN" ] || { echo "STOPPED: not signed in to gcloud. Nothing changed."; exit 1; }
export P MODE TOKEN
python3 - <<'PY'
import json, os, sys, time, urllib.request, urllib.error
P, MODE, TOKEN = os.environ["P"], os.environ["MODE"], os.environ["TOKEN"]
FSB = os.environ.get("GST_FS", "https://firestore.googleapis.com/v1")
AUB = os.environ.get("GST_AUTH", "https://identitytoolkit.googleapis.com/v1")
H = {"Authorization": "Bearer " + TOKEN, "Content-Type": "application/json", "x-goog-user-project": P}
stamp = time.strftime("%Y%m%d-%H%M%S")
LIST = f"gst-list-{stamp}.txt"
out = open(LIST, "w")
def note(line=""): out.write(line + "\n")
def stop(msg):
    out.close()
    print(f"\nSTOPPED. Nothing was changed.\nWhy: {msg}\nThe details so far are in {LIST}"); sys.exit(1)
def call(url, body=None):
    req = urllib.request.Request(url, data=json.dumps(body).encode() if body is not None else None, headers=H, method="POST" if body is not None else "GET")
    try:
        with urllib.request.urlopen(req) as r: return json.loads(r.read() or b"{}")
    except urllib.error.HTTPError as e:
        stop(f"Google answered {e.code}: {e.read()[:200].decode(errors='replace')}")
    except Exception as e:
        stop(f"no answer from Google ({e})")
FS = f"{FSB}/projects/{P}/databases/(default)/documents"
print("Working… (reading sign-ins and groups; this takes about a minute)")
# every sign-in
users, tok = [], ""
while True:
    j = call(f"{AUB}/projects/{P}/accounts:batchGet?maxResults=1000" + (f"&nextPageToken={tok}" if tok else ""))
    users += j.get("users", []); tok = j.get("nextPageToken", "")
    if not tok: break
# every membership in every group
members = set()
for d in call(f"{FS}:runQuery", {"structuredQuery": {"from": [{"collectionId": "members", "allDescendants": True}], "select": {"fields": []}}}):
    if isinstance(d, dict) and "document" in d: members.add(d["document"]["name"].split("/")[-1])
now = time.time() * 1000
day = lambda ms: time.strftime("%Y-%m-%d", time.gmtime(int(ms) / 1000)) if ms else "never"
guests = [u for u in users if not u.get("email") and not u.get("providerUserInfo")]
old = [u for u in guests if u["localId"] not in members and now - int(u.get("lastLoginAt") or u.get("createdAt") or 0) > 30 * 86400000]
# Safety (Willy, 6 Nov: no data corruption): a guest that any golfer is linked
# to, or that owns a group, is KEPT, so no golfer or group points at a removed
# sign-in.
def everyDoc(coll):
    res, t = [], ""
    while True:
        j = call(f"{FS}/{coll}?pageSize=300" + (f"&pageToken={t}" if t else ""))
        res += j.get("documents", []); t = j.get("nextPageToken", "")
        if not t: return res
sv = lambda d, k: ((d.get("fields") or {}).get(k) or {}).get("stringValue", "")
linked = {sv(d, "linkedUid") for d in everyDoc("golfers")} - {""}
owners = {sv(d, "ownerUid") for d in everyDoc("associations")} - {""}
kept = [u for u in old if u["localId"] in linked or u["localId"] in owners]
old = [u for u in old if u not in kept]
note(f"STEP 13 — guest sign-ins in no group, unused 30+ days, linked to nothing: {len(old)}")
note(f"kept because a golfer or group points at them: {len(kept)}")
for u in kept: note(f"  KEPT {u['localId']}")
for u in old: note(f"  {u['localId']}  made {day(u.get('createdAt'))}  last used {day(u.get('lastLoginAt'))}")
docs = []
for u in old:
    for parent in (f"users/{u['localId']}/terms", f"users/{u['localId']}/prefs", f"userGroups/{u['localId']}/groups"):
        docs += [(d["name"], d["updateTime"]) for d in call(f"{FS}/{parent}?pageSize=300").get("documents", [])]
note(f"their own small records: {len(docs)}")
for n, _ in docs: note("  " + n.split("/documents/")[1])
# step 29
have, tok = set(), ""
while True:
    j = call(f"{FS}/accountEmails?pageSize=300" + (f"&pageToken={tok}" if tok else ""))
    have |= {d["name"].split("/")[-1] for d in j.get("documents", [])}
    tok = j.get("nextPageToken", "")
    if not tok: break
withEmail = [u for u in users if u.get("email")]
missing = [u for u in withEmail if u["email"].strip().lower() not in have]
note(f"\nSTEP 29 — accounts with an email: {len(withEmail)}; missing from the account-email list: {len(missing)}")
for u in missing: note("  " + u["email"])
out.close()

def summary():
    print("\n================ SUMMARY ================")
    print(f"Sign-ins in total:                       {len(users)}")
    print(f"Guest sign-ins (no email):               {len(guests)}")
    print(f"  in no group, unused 30+ days:          {len(old) + len(kept)}")
    print(f"  kept (a golfer or group points at it): {len(kept)}")
    print(f"  to be removed:                         {len(old)}")
    print(f"  their own small records:               {len(docs)}")
    print(f"Accounts with an email:                  {len(withEmail)}")
    print(f"  missing from the account-email list:   {len(missing)}")
    for u in missing[:10]: print(f"     {u['email']}")
    print(f"The full list is in the file {LIST}")
if MODE != "go":
    summary()
    print("\nLIST ONLY: nothing was changed.")
    print("To do exactly this, type:  bash gst.txt go")
    sys.exit(0)

if not old and not missing:
    summary(); print("\nNothing to do. Nothing was changed."); sys.exit(0)
backup = f"gst-backup-{stamp}.json"
json.dump({"guestSignIns": old, "records": [call(f"{FSB}/{n}") for n, _ in docs], "missingEmails": [u["email"] for u in missing]}, open(backup, "w"))
done = []
if docs:
    if len(docs) > 500: stop("more than 500 records; ask Claude for a split run")
    r = call(f"{FS}:commit", {"writes": [{"delete": n, "currentDocument": {"updateTime": t}} for n, t in docs]})
    done.append(f"guests' small records removed in one commit: {len(docs)}")
removed, notRemoved = 0, []
ids = [u["localId"] for u in old]
for i in range(0, len(ids), 1000):
    r = call(f"{AUB}/projects/{P}/accounts:batchDelete", {"localIds": ids[i:i+1000], "force": True})
    errs = r.get("errors", []); notRemoved += errs; removed += len(ids[i:i+1000]) - len(errs)
if ids: done.append(f"guest sign-ins removed: {removed}" + (f"; NOT removed: {len(notRemoved)}" if notRemoved else ""))
if missing:
    at = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    call(f"{FS}:commit", {"writes": [{"update": {"name": f"projects/{P}/databases/(default)/documents/accountEmails/{u['email'].strip().lower()}", "fields": {"at": {"timestampValue": at}}}, "currentDocument": {"exists": False}} for u in missing]})
    done.append(f"account-email entries added in one commit: {len(missing)}")
summary()
print(f"\nBackup saved: {backup}")
print("DONE:"); [print("  " + d) for d in done]
PY
